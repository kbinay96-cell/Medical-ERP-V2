-- =========================================================
-- Medical ERP V2 - Module 01: Login & Dashboard
-- Authentication / Authorization Schema
-- =========================================================

-- ---------------------------------------------------------
-- COMPANY (minimal, needed for cmbCompany on Login screen)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS company (
    companyid       VARCHAR(20) PRIMARY KEY,
    companyname     VARCHAR(200) NOT NULL,
    country         VARCHAR(50) NOT NULL DEFAULT 'Nepal',
    status          VARCHAR(20) NOT NULL DEFAULT 'Active',
    createddate     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ---------------------------------------------------------
-- FINANCIAL YEAR (minimal, needed for cmbFinancialYear)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS financialyear (
    financialyearid SERIAL PRIMARY KEY,
    financialyear   VARCHAR(20) NOT NULL,
    startbsdate     VARCHAR(10) NOT NULL,
    endbsdate       VARCHAR(10) NOT NULL,
    startaddate     DATE,
    endaddate       DATE,
    isactive        BOOLEAN NOT NULL DEFAULT FALSE,
    isclosed        BOOLEAN NOT NULL DEFAULT FALSE,
    createdat       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ---------------------------------------------------------
-- ROLES  (RBAC)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS roles (
    roleid          SERIAL PRIMARY KEY,
    rolename        VARCHAR(50) NOT NULL UNIQUE,
    description     VARCHAR(200),
    status          VARCHAR(20) NOT NULL DEFAULT 'Active',
    createddate     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ---------------------------------------------------------
-- ROLE PERMISSIONS (per screen, per action)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS role_permissions (
    permissionid    SERIAL PRIMARY KEY,
    roleid          INTEGER NOT NULL REFERENCES roles(roleid),
    screenname      VARCHAR(100) NOT NULL,
    can_view        BOOLEAN NOT NULL DEFAULT FALSE,
    can_add         BOOLEAN NOT NULL DEFAULT FALSE,
    can_edit        BOOLEAN NOT NULL DEFAULT FALSE,
    can_delete      BOOLEAN NOT NULL DEFAULT FALSE,
    can_restore     BOOLEAN NOT NULL DEFAULT FALSE,
    can_print       BOOLEAN NOT NULL DEFAULT FALSE,
    can_export      BOOLEAN NOT NULL DEFAULT FALSE,
    can_import      BOOLEAN NOT NULL DEFAULT FALSE,
    can_approve     BOOLEAN NOT NULL DEFAULT FALSE,
    can_cancel      BOOLEAN NOT NULL DEFAULT FALSE,
    can_lock        BOOLEAN NOT NULL DEFAULT FALSE,
    can_unlock      BOOLEAN NOT NULL DEFAULT FALSE,
    UNIQUE (roleid, screenname)
);

-- ---------------------------------------------------------
-- USERS
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS users (
    userid              SERIAL PRIMARY KEY,
    username            VARCHAR(50) NOT NULL UNIQUE,
    passwordhash        VARCHAR(256) NOT NULL,
    passwordsalt        VARCHAR(64) NOT NULL,
    fullname            VARCHAR(150),
    roleid              INTEGER NOT NULL REFERENCES roles(roleid),
    companyid           VARCHAR(20) REFERENCES company(companyid),
    status              VARCHAR(20) NOT NULL DEFAULT 'Active',
        -- Active | Disabled | Locked | Expired | Suspended | Deleted
    failedattempts      INTEGER NOT NULL DEFAULT 0,
    lockeduntil         TIMESTAMP,
    passwordchangedat   TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    mustchangepassword  BOOLEAN NOT NULL DEFAULT FALSE,
    remembertoken       VARCHAR(256),
    remembertokenexpiry TIMESTAMP,
    createddate         TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    createdby           VARCHAR(50),
    modifieddate        TIMESTAMP,
    modifiedby          VARCHAR(50)
);

-- ---------------------------------------------------------
-- LOGIN HISTORY (successful logins)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS login_history (
    loginid         BIGSERIAL PRIMARY KEY,
    userid          INTEGER NOT NULL REFERENCES users(userid),
    username        VARCHAR(50) NOT NULL,
    companyid       VARCHAR(20),
    financialyear   VARCHAR(20),
    logintime       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    logouttime      TIMESTAMP,
    machinename     VARCHAR(100),
    appversion      VARCHAR(20),
    sessionid       VARCHAR(64) NOT NULL
);

-- ---------------------------------------------------------
-- FAILED LOGIN HISTORY
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS failed_login_history (
    failedid        BIGSERIAL PRIMARY KEY,
    username        VARCHAR(50) NOT NULL,
    attemptdate     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    machinename     VARCHAR(100),
    failurereason   VARCHAR(200)
);

-- ---------------------------------------------------------
-- AUDIT LOG (never editable / never deleted by app code)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS audit_log (
    auditid         BIGSERIAL PRIMARY KEY,
    userid          INTEGER REFERENCES users(userid),
    username        VARCHAR(50),
    action          VARCHAR(100) NOT NULL,
        -- Login | Logout | PasswordChanged | PasswordReset |
        -- UserLocked | UserUnlocked | RoleChanged | PermissionChanged
    details         TEXT,
    createdat       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ---------------------------------------------------------
-- SESSIONS (active/recent sessions)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS sessions (
    sessionid       VARCHAR(64) PRIMARY KEY,
    userid          INTEGER NOT NULL REFERENCES users(userid),
    companyid       VARCHAR(20),
    financialyear   VARCHAR(20),
    logintime       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    lastactivity    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    machinename     VARCHAR(100),
    appversion      VARCHAR(20),
    isactive        BOOLEAN NOT NULL DEFAULT TRUE
);

-- ---------------------------------------------------------
-- LICENSE
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS license (
    licenseid       SERIAL PRIMARY KEY,
    licensekey      VARCHAR(100) NOT NULL UNIQUE,
    machineid       VARCHAR(100),
    productedition  VARCHAR(50) NOT NULL DEFAULT 'Standard',
    enabledmodules  TEXT,               -- comma separated module list
    issuedate       DATE NOT NULL DEFAULT CURRENT_DATE,
    expirydate      DATE,
    status          VARCHAR(20) NOT NULL DEFAULT 'Active'
);

-- ---------------------------------------------------------
-- SUBSCRIPTION
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS subscription (
    subscriptionid  SERIAL PRIMARY KEY,
    plan            VARCHAR(50) NOT NULL DEFAULT 'Standard',
    startdate       DATE NOT NULL DEFAULT CURRENT_DATE,
    enddate         DATE,
    gracedays       INTEGER NOT NULL DEFAULT 7,
    maxusers        INTEGER NOT NULL DEFAULT 5,
    maxbranches     INTEGER NOT NULL DEFAULT 1,
    status          VARCHAR(20) NOT NULL DEFAULT 'Active'
);

-- ---------------------------------------------------------
-- SETTINGS (generic key-value, same pattern as V1)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS settings (
    setting_id      SERIAL PRIMARY KEY,
    setting_key     VARCHAR(150) NOT NULL,
    setting_value   TEXT,
    setting_group   VARCHAR(50) NOT NULL,
    data_type       VARCHAR(20) NOT NULL DEFAULT 'string',
        -- string | integer | decimal | boolean | date | time | json | color | font | path | printer
    default_value   TEXT,
    description     TEXT,
    is_editable     BOOLEAN NOT NULL DEFAULT TRUE,
    display_order   INTEGER NOT NULL DEFAULT 0,
    companyid       VARCHAR(20) REFERENCES company(companyid),
        -- NULL = global setting (applies to every company)
    userid          INTEGER REFERENCES users(userid),
        -- NULL = global/company-level setting (not a per-user override)
    created_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP,
    updated_by      VARCHAR(50)
);

-- A plain UNIQUE(setting_key, companyid, userid) constraint would NOT
-- actually stop duplicate global settings, because PostgreSQL treats
-- every NULL as distinct from every other NULL. A partial unique index
-- is required to enforce "only one global row per setting_key".
CREATE UNIQUE INDEX IF NOT EXISTS idx_settings_global_unique
    ON settings(setting_key) WHERE companyid IS NULL AND userid IS NULL;

-- Company-level override: one row per (setting_key, companyid) when userid is NULL.
CREATE UNIQUE INDEX IF NOT EXISTS idx_settings_company_unique
    ON settings(setting_key, companyid) WHERE companyid IS NOT NULL AND userid IS NULL;

-- User-level override: one row per (setting_key, companyid, userid) when userid is set.
CREATE UNIQUE INDEX IF NOT EXISTS idx_settings_user_unique
    ON settings(setting_key, companyid, userid) WHERE userid IS NOT NULL;

CREATE INDEX IF NOT EXISTS idx_settings_key ON settings(setting_key);
CREATE INDEX IF NOT EXISTS idx_settings_group ON settings(setting_group);

-- ---------------------------------------------------------
-- SETTINGS HISTORY (every save creates a permanent record -
-- never editable/deletable by the application)
-- ---------------------------------------------------------
CREATE TABLE IF NOT EXISTS settings_history (
    history_id      BIGSERIAL PRIMARY KEY,
    setting_key     VARCHAR(150) NOT NULL,
    old_value       TEXT,
    new_value       TEXT,
    changed_by      VARCHAR(50),
    changed_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    reason          TEXT
);
